using System;
using System.Collections.Generic;

namespace ExpenseTracking
{
    class Expense
    {
        public string Category { get; set; }
        public double Amount { get; set; }
        public DateTime Date { get; set; }
    }

    class Program
    {
        static List<Expense> expenses = new List<Expense>();

        static void Main(string[] args)
        {
            while (true)
            {
                Console.WriteLine("\n===== Expense Tracking System =====");
                Console.WriteLine("1. Add Expense");
                Console.WriteLine("2. View Expenses");
                Console.WriteLine("3. Total Expense");
                Console.WriteLine("4. Exit");
                Console.Write("Enter your choice: ");

                try
                {
                    int choice = Convert.ToInt32(Console.ReadLine());

                    switch (choice)
                    {
                        case 1:
                            AddExpense();
                            break;
                        case 2:
                            ViewExpenses();
                            break;
                        case 3:
                            TotalExpense();
                            break;
                        case 4:
                            Console.WriteLine("Thank You ");
                            return;
                        default:
                            Console.WriteLine("Invalid Choice");
                            break;
                    }
                }
                catch (FormatException)
                {
                    Console.WriteLine("Error: Please enter a valid number");
                }
                catch (Exception ex)
                {
                    Console.WriteLine("Unexpected Error: " + ex.Message);
                }
            }
        }

        static void AddExpense()
        {
            try
            {
                Expense exp = new Expense();

                Console.Write("Enter Category: ");
                exp.Category = Console.ReadLine();

                Console.Write("Enter Amount: ");
                exp.Amount = Convert.ToDouble(Console.ReadLine());

                if (exp.Amount <= 0)
                    throw new Exception("Amount must be greater than zero.");

                exp.Date = DateTime.Now;

                expenses.Add(exp);

                Console.WriteLine("Expense Added Successfully.");
            }
            catch (FormatException)
            {
                Console.WriteLine("Invalid amount. Please enter numeric value");
            }
            catch (Exception ex)
            {
                Console.WriteLine("Error: " + ex.Message);
            }
        }

        static void ViewExpenses()
        {
            if (expenses.Count == 0)
            {
                Console.WriteLine("No expenses found.");
                return;
            }

            Console.WriteLine("\nCategory\tAmount\tDate");

            foreach (Expense exp in expenses)
            {
                Console.WriteLine($"{exp.Category}\t\t{exp.Amount}\t{exp.Date.ToShortDateString()}");
            }
        }

        static void TotalExpense()
        {
            double total = 0;

            foreach (Expense exp in expenses)
            {
                total += exp.Amount;
            }

            Console.WriteLine("Total Expense = " + total);
        }
    }
}
